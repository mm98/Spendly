	/** The optional link band across the top. Omit it and the card opens straight on the
	 *  headline. `progress` draws the scan glyph beside the title; `destination` is spoken
	 *  after the visible text as ", open <destination>". */
	header?: { to: string; title: string; meta?: string; destination?: string; progress?: boolean };
	/** The bold line the eye lands on. */
	headline?: string;
	/** The sentence under the headline. */
	body?: React.ReactNode;
	/** The action row at the foot. Pass `<Button variant="invert">`, `<ButtonLink
	 *  className="button-link-invert">` and `<AiCardMark />` - the card is dark, so a
	 *  default-styled button disappears into it. */
	children?: React.ReactNode;
