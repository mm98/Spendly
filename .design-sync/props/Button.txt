	/** Space-separated variant words, each becoming a `button-<word>` class:

	variant?: string;

	to?: string;

	href?: string;
	/** An Icon name placed before the label. */
	icon?: string;
	/** An Icon name placed after the label. */
	iconAfter?: string;

	children?: React.ReactNode;
	className?: string;

	type?: "button" | "submit" | "reset";
	disabled?: boolean;
