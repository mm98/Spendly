	/** The glyph's name. A key of `ICONS` draws on its own; a Tabler name needs `iconPath`
	 *  beside it. */
	icon?: string;
	/** The SVG path the record stored when the icon was chosen. It wins over `icon`, and it is
	 *  why a chosen glyph draws without loading the icon catalogue. */
	iconPath?: string;
	/** The tile's fill, as a hex string. The glyph's own ink is picked from it, so a pale fill
	 *  still shows its glyph. */
	colour?: string;
	/** The category's name. Empty draws "Your category", so the row keeps its shape while the
	 *  name is still being typed. */
	name?: string;
