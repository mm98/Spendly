	/** The category record. `colour` and `icon` draw the tile, `name` the title, `spend` the
	 *  figure (formatted as whole kroner), and `share` the default meta line. `iconPath` is
	 *  a custom glyph path where the record carries one instead of a named icon. */
	category?: {
		id?: string;
		name: string;
		colour: string;
		icon: string;
		iconPath?: string | null;
		/** Percent of expenses, e.g. 38.4 — written by the row as "38,4% of expenses". */
		share?: number;
		/** Rendered in the account's currency, rounded to whole units. */
		spend: number;
	};
	/** Replaces the "38,4% of expenses" line the row writes for itself — e.g. what the spend
	 *  is against a budget. Say what stretch of time `spend` covers if it is not all time. */
	meta?: string;
	/** An in-app route. With it the whole row is a link and draws a chevron. */
	to?: string;
