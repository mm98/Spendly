	/** The chosen colour, as a hex string (`#14AC72`). A `var(--token)` reference is read off
	 *  the stylesheet, but records store the hex. A value outside `colours` rings no swatch and
	 *  fills the last box instead. */
	value?: string;
	/** Gets the chosen colour as a hex string, from a swatch or from the platform picker. */
	onChange?: (colour: string) => void;
	/** The swatches on offer. Defaults to `CATEGORY_COLOURS`; pass `VENDOR_COLOURS` for a
	 *  vendor, whose tile carries two white letters and needs the darker end. Both are exported
	 *  beside this component. Eight is the row the layout is drawn for. */
	colours?: Array<{ id: string; label: string }>;
	/** The accessible name of the whole row, e.g. "Category colour". */
	label?: string;
