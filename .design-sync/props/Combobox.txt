	value?: string;

	/** Sections of the open list. */
	groups?: Array<{ heading?: string; options: Array<{ id: string; label: string }> }>;

	action?: string;

	invalid?: boolean;

	leadingIcon?: string;

	chips?: React.ReactNode;

	trail?: React.ReactNode;

	pickOnly?: boolean;
