	/** The question, ending in a question mark: "Delete this order?". Also the dialog's
	 *  accessible name. */
	title?: string;
	/** One sentence naming what is at stake, so the reader can tell they picked the right
	 *  record. `useDelete` appends "This cannot be undone." to it. */
	detail?: string;
	/** The verb on the destructive button, in capitals: "DELETE", "RELOAD". */
	confirmLabel?: string;
	/** The safe answer. Defaults to "CANCEL". */
	cancelLabel?: string;
	/** Runs the action. The caller unmounts the dialog; it does not close itself. */
	onConfirm?: () => void;
	/** Backdrop click and Escape both call this, as well as the cancel button. */
	onCancel?: () => void;
