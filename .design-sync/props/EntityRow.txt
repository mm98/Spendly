	/** The record's own tile, passed in whole: a vendor's initials avatar or a category's
	 *  coloured glyph, usually inside an `EntityMark` so a favourite carries its star. */
	mark?: React.ReactNode;
	/** The record's name — the row's first line, and the link when the row goes somewhere. */
	name?: string;
	/** The quiet second line: order count, share, budget, a matched alias. */
	meta?: string;
	/** The figure on the right, already formatted. */
	amount?: React.ReactNode;
	/** A smaller second figure under `amount`, e.g. "saved 1.965 kr.". */
	sub?: string;
	/** An in-app route — the record's own edit page. Without it, and without `open`, the row
	 *  is inert: no link, no chevron. */
	editTo?: string;
	/** A second control riding on top of the row, e.g. an edit IconButton. Only meaningful
	 *  beside `open` or `editTo`, since one link cannot hold another. */
	action?: React.ReactNode;
	/** Sends the row itself out of the app, in a new tab. The row draws no chevron then —
	 *  it is leaving. `label` is what the link announces. */
	open?: { href: string; label: string } | null;
