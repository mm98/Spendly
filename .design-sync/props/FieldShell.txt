	/** Extra class on the shell:
	 *  `field-shell-area` (taller, for a textarea) | `field-shell-date` |
	 *  `field-shell-chips` (wraps chips onto more lines) |
	 *  `field-shell-no-chevron` (with `field-shell-chips`, when no list opens). */
	variant?: string;
	/** The control and anything riding inside the border with it: an input, an Icon, a
	 *  FieldUnit, Chips. */
	children?: React.ReactNode;
	className?: string;
	/** The element to render. `div` by default; `label` makes the whole shell clickable,
	 *  which is what the file picker uses. */
	as?: "div" | "label" | "span";
	[key: string]: unknown;
