	/** The card's headline, and its accessible name unless `label` says otherwise. */
	title?: string;
	/** One sentence under the headline saying what the form is for. */
	lead?: string;
	/** Overrides the card's aria-label. Defaults to `title`. */
	label?: string;
	/** Draws the step banner above the card. `done` puts a check on a step behind you, `on`
	 *  marks the one you are on. Leave it out for a form that is not a wizard. */
	steps?: Array<{ label: string; icon: string; done?: boolean; on?: boolean }>;
	/** A verdict about the whole form, drawn between the lead and the first field - normally a
	 *  `Note`. Never a rejection of one field: that belongs under the field. */
	notice?: React.ReactNode;
	/** The column beside the form: the live summary card, and on an edit form a `DeleteCard`.
	 *  Pass a fragment for more than one card. */
	rail?: React.ReactNode;
	/** The left button of the footer, described rather than built - the page draws it as a
	 *  ghost button. `label` defaults to "CANCEL". */
	cancel?: { to?: string; onClick?: () => void; label?: string; icon?: string };
	/** The right button of the footer, drawn as `variant="caps"`. Write the label in capitals
	 *  and swap it for "SAVING..." while the request is in flight. */
	submit?: { label: string; onClick: () => void; disabled?: boolean };
	/** The fields, in order: `Field`, `FieldPair`, `FilterSection`. The footer is drawn for
	 *  you, so never pass a `FormActions` of your own. */
	children?: React.ReactNode;
