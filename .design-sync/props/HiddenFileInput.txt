	/** Gets the chosen files as an array. Fired only when something was chosen - cancelling the
	 *  dialog is an outcome, not a call. */
	onFiles?: (files: File[]) => void;
	/** Whether the dialog allows more than one file. Default `false`. */
	multiple?: boolean;
	/** What the dialog filters to, from `fileKinds.js`: `RECEIPT_FILES`, `PHOTO_FILES`,
	 *  `EXPORT_FILES`. Shaped `{description, accept}`. With it, browsers that have
	 *  showOpenFilePicker leave the "All files" entry out of the dialog. */
	kind?: { description: string; accept: Record<string, string[]> };
	/** An `accept` attribute written out by hand, for an input with no `kind`. */
	accept?: string;
	/** The input's accessible name, e.g. "Choose a JSON file of orders". */
	label?: string;
	/** The handle the opener holds. `ref.current.click()` opens the best dialog the browser
	 *  offers; it is not the input element itself. */
	ref?: React.Ref<{ click: () => void }>;
