	/** A key of `ICONS`, which is exported beside this. An unknown name throws rather than
	 *  drawing nothing. */
	name?: string;
	/** An SVG path `d` drawn instead of a named glyph, and it wins over `name`. This is how a
	 *  record that stored its own icon is drawn without loading a catalogue. */
	path?: string;
	/** Defaults to `"icon"`, which is the class that sizes and strokes it. Pass `""` to opt out
	 *  of that (the brand mark does). */
	className?: string;
	/** Makes it a labelled image. Without one the svg is `aria-hidden`, which is right for a
	 *  glyph sitting beside its own words. */
	title?: string;
	[key: string]: unknown;
