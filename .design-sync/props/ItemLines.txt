	/** The order's lines, in the order they are drawn. `retail` and `note` are optional; `id`
	 *  is any stable string (a new line gets one of its own). */
	items?: Array<{ id: string; title: string; quantity: number; price: number;
		retail?: number; note?: string }>;
	/** Gets the whole new array, on every add, edit, delete and reorder. The host owns the
	 *  list; this component holds no copy of it. */
	onItemsChange?: (items: unknown[]) => void;
	/** A sentence to announce: a line saved, removed, or moved. Pass the page's toast so the
	 *  screen has one live region rather than two. */
	onAnnounce?: (message: string) => void;
	/** The rejection for the list as a whole, e.g. "An order needs at least one item." Drawn
	 *  where a field's own message would be. Set it only after a submit. */
	error?: string;
	/** What the host knows about one line's picture: `{present, src}`, or `{}`. It answers from
	 *  whatever it has - a file picked this session, an attachment already on the server. */
	photoOf?: (item: unknown) => { present?: boolean; src?: string };
	/** A picture was chosen for a line. The id is the line's, and may be a brand new one. */
	onPhotoPick?: (id: string, file: File) => void;
	/** A line's picture was removed, or the line itself was. */
	onPhotoClear?: (id: string) => void;
	/** The word the list calls itself, in the heading row and in what is announced. Default
	 *  "Items". */
	label?: string;
	/** Draws the row above the list holding the name, the count and the hint. Default `true`.
	 *  Pass `false` where the card's own header band already carries the first two. */
	heading?: boolean;
