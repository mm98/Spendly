	/** What the figure measures, e.g. "Spent". */
	label?: string;
	/** A number, rendered as currency in the account's currency. Use instead of `value`. */
	amount?: number;
	/** An already-formatted string, for counts. Use instead of `amount`. */
	value?: string;
	/** The comparison sentence, e.g. "8% more than August". Omit when nothing to compare against. */
	delta?: string;
	/** Picks the arrow beside `delta`. `level` draws none. */
	trend?: "up" | "down" | "level";
	/** Picks the colour of `delta`. Independent of `trend`: spending less is a good `down`. */
	tone?: "good" | "bad" | "neutral";
	/** A quiet line under the figure. */
	note?: string;
