	/** What the orders are being moved between, singular and lowercase:
	 *  `"vendor"` or `"category"`. It writes every label on the card. */
	what?: string;
	/** The name they are moving away from, named in the confirming question. */
	from?: string;
	/** How many orders would move. Zero draws the card with its sentence and no
	 *  control, since there is nothing to move. */
	count?: number;
	/** The list to choose from, in the Combobox's own shape. */
	groups?: Array<{ heading?: string; options: Array<{ id: string; label: string }> }>;
	/** Called with the chosen option once the reader confirms. Return a promise and
	 *  the button says MOVING... until it settles. */
	onMove?: (target: { id: string; label: string }) => void | Promise<unknown>;
	/** The sentence under the heading, in the muted lead size. */
	children?: React.ReactNode;
