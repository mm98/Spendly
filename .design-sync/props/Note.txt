	/** `positive` (green) or `warning` (amber). Defaults to `positive`. These are Note's own
	 *  words - KpiCard's `good | bad | neutral` are a different set and render nothing here. */
	tone?: "positive" | "warning";
	/** The verdict, inked in the tone's colour and set bold at the start of the line. */
	title?: string;
	/** The sentence after the title, in the same paragraph. */
	children?: React.ReactNode;
	/** `note-lead` for a note that opens a card, `note-above-row` for one sitting directly
	 *  over the row it is about. */
	className?: string;
	[key: string]: unknown;
