	/** The figure as text, not a number: Danish decimals use a comma, and
	 *  `input[type=number]` hands back an empty value for one. */
	value?: string;
	/** Gets the typed figure, filtered to what a save will accept. */
	onChange?: (value: string) => void;
	/** The short suffix drawn inside the field, e.g. `kr.` or `pcs`. Left out, the
	 *  field is the figure alone. */
	unit?: React.ReactNode;
	/** The control's id, and what the Field's `htmlFor` points at. Generated when
	 *  left out, but then no label or error can address the field. */
	id?: string;
	/** Draws the red edge and sets aria-invalid. */
	invalid?: boolean;
	/** The Field's `errorId`. Only read while `invalid` is set. */
	describedBy?: string;
	placeholder?: string;
	[key: string]: unknown;
