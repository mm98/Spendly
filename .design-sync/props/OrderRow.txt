	/** The whole order record, in the shape the orders list endpoint returns. `dateLabel` is
	 *  the one field the page adds: the API sends `date` as an ISO day and the page formats
	 *  it. `total` and `saved` are numbers and the row formats them; `saved` is drawn even
	 *  when it is 0, so the total column keeps one shape down the list. */
	order?: {
		id: string;
		/** ISO day, e.g. "2026-09-02". */
		date?: string;
		/** The formatted date the row prints, e.g. "2 Sep". */
		dateLabel: string;
		status: "Ordered" | "Delivered" | "Returned";
		vendor: { id: string; name: string; initials: string; colour: string } | null;
		/** Named on the meta line, comma-separated. Empty array for an order with none. */
		categories: Array<{ id: string; name: string; colour: string; icon: string }>;
		itemCount: number;
		total: number;
		saved: number;
	};
	/** An in-app route — the order's own page. */
	to?: string;
	[key: string]: unknown;
