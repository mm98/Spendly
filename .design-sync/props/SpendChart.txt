	/** The plotted points, `[{ x, y1, y2?, date? }]`. `x` is a **day offset from `from`**, not
	 *  an array index: a window drawn one point per day is 0,1,2..., and a window cut into
	 *  months carries each month's own first day (0, 31, 59, ...) so the axis marks line up
	 *  with the points. `y1` is the first series, `y2` the second where there is one. */
	data: { x: number; y1: number; y2?: number }[];
	/** The window's real day span, so the axis runs start to end even when only the first few
	 *  days have a point yet. The x domain is `[0, days - 1]`. */
	days: number;
	/** The window's first day, ISO `YYYY-MM-DD`. Only the tooltip's date label uses it. */
	from: string;
	/** The axis marks: one entry per label, each carrying the day it belongs to. `label` is
	 *  what is drawn; `fullLabel` is the unabbreviated form and is not drawn by this chart. */
	marks: { label: string; fullLabel?: string; day: number }[];
	/** One entry per line drawn. `dataKey` picks the field off `data`. `area` fills under the
	 *  line with `color` faded to nothing; `strokeGradient` paints the line itself left to
	 *  right with a `[from, to]` colour pair (`color` still marks the tooltip swatch and the
	 *  today dot); `dash` is an SVG dash pattern such as `'4 4'`; `today` draws the "you are
	 *  here" dot, and only where `todayIndex` resolves to a real point. */
	series: {
		key: string;
		dataKey: 'y1' | 'y2';
		name: string;
		color: string;
		area?: boolean;
		strokeGradient?: [string, string];
		dash?: string;
		strokeWidth?: number;
		today?: boolean;
	}[];
	/** Which point of `data` is today, or `null` where the window has no today - a window that
	 *  has ended, and a comparison series regardless. */
	todayIndex?: number | null;
	/** Formatter for the value axis's labels. Give it the chart shorthand (60000 -> "60k"),
	 *  not a full currency string - the labels repeat down the edge. */
	yFormat?: (value: number) => string;
	/** Plot height in pixels. The x-axis's own tick row is added on top of it, so the box is
	 *  taller than this by 20px. */
	height: number;
	/** The chart's accessible name. It is one `role="img"`, so say what is plotted, over what
	 *  stretch, and in what unit. */
	label?: string;
	/** Extra class on the chart's own container - `chart-under-caption` where it sits under a
	 *  `chart-caption` block. */
	className?: string;
