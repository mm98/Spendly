	/** The arcs, in draw order clockwise from twelve o'clock. `share` is a percentage and
	 *  `colour` a CSS colour - the category's own, so the ring and its key agree. Omitted or
	 *  empty, the ring draws no arcs: a flat track with the centre figure still in it. */
	segments?: { share: number; colour: string }[];
	/** The figure in the hole, already formatted (`"12.474 kr."`). It shrinks itself to
	 *  whatever the circle can hold. */
	value?: string;
	/** The small caption under the figure - usually the window's name. */
	label?: string;
	/** The whole ring's accessible name; Recharts' own SVG stays hidden behind it. */
	ariaLabel?: string;
