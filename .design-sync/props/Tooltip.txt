	/** The rectangle of what is being pointed at, from `getBoundingClientRect()`.
	    Nothing drawn without one -- that is how the label is hidden. */
	anchor?: DOMRect;
	/** Optional heading line above the body, in the muted caption size. */
	title?: React.ReactNode;
	/** The body. Plain text reads as normal weight; compose rows from
	    `tooltip-row` / `tooltip-swatch` / `tooltip-name` / `tooltip-value` for figures. */
	children?: React.ReactNode;
