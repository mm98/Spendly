	/** Seven entries, Monday first. `share` is the bar's height as a percentage of the busiest
	 *  day (0-100), not a pixel figure; `spend` is the real amount and is drawn only in the
	 *  tooltip; `peak` marks the one bar filled green. `day` is the short tick label and
	 *  `dayFull` the long one - both travel on every tick and the stylesheet picks one. */
	days: { day: string; dayFull?: string; spend: number; share: number; peak?: boolean }[];
	/** `rhythm-bars` makes the chart its own container, so the ticks print full day names
	 *  wherever there is room for them. Without it the short form is always drawn. */
	className?: string;
	/** Inline style on the chart's own box, e.g. to override its height. */
	style?: React.CSSProperties;
	/** Given, the whole chart becomes one labelled image. Left out, the chart is hidden from
	 *  screen readers - which is right only where the same facts are stated in words beside
	 *  it, as on Home's rhythm card. */
	ariaLabel?: string;
